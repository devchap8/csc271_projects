// Downloads the image for every card in cards.json into ./images/
// Run from the product_site folder:  node download-images.js
// Needs Node 18+ (built-in fetch). Images come from Wikimedia Commons,
// where every file is free to use under its listed license.

const fs = require("fs");
const path = require("path");

const cards = JSON.parse(fs.readFileSync(path.join(__dirname, "cards.json"), "utf8"));
const WIDTH = 800; // resize on Wikimedia's side so files stay small

async function download(card) {
  // "https://commons.wikimedia.org/wiki/File:Cute_dog.jpg" -> "Cute_dog.jpg"
  const fileName = decodeURIComponent(card.imageSource.split("/wiki/File:")[1]);
  const url = `https://commons.wikimedia.org/wiki/Special:FilePath/${encodeURIComponent(fileName)}?width=${WIDTH}`;

  // Wikimedia asks scripts to send a descriptive User-Agent
  const res = await fetch(url, { headers: { "User-Agent": "CardWellSchoolProject/1.0" } });
  if (!res.ok) throw new Error(`${res.status} for ${fileName}`);

  const outPath = path.join(__dirname, card.image);
  fs.mkdirSync(path.dirname(outPath), { recursive: true });
  fs.writeFileSync(outPath, Buffer.from(await res.arrayBuffer()));
  console.log(`saved ${card.image}`);
}

(async () => {
  for (const card of cards) {
    try {
      await download(card);
    } catch (err) {
      console.error(`FAILED card ${card.id} (${card.word}): ${err.message}`);
    }
  }
})();
